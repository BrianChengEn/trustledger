Internal OpenAPI

openapi: 3.1.0
info:
  title: TrustLedger Internal API
  version: 0.1.0-mvp
  description: >-
    Team-owned React-to-FastAPI contract for the Technica 2026 B2C MVP.
    This is not a Capital One, Jev, or Backboard schema.
servers:
  - url: http://127.0.0.1:8000
paths:
  /api/v1/health:
    get:
      operationId: getHealth
      responses:
        '200':
          description: Safe readiness status. Never includes keys or upstream bodies.
          content:
            application/json:
              schema:
                $ref: '#/components/schemas/HealthResponse'
  /api/v1/transactions:
    get:
      operationId: listTransactions
      parameters:
        - name: limit
          in: query
          schema: { type: integer, minimum: 1, maximum: 100, default: 25 }
      responses:
        '200':
          description: Canonical transactions for the current demo user.
          content:
            application/json:
              schema:
                type: array
                items: { $ref: '#/components/schemas/Transaction' }
  /api/v1/analyses:
    post:
      operationId: analyzeTransaction
      requestBody:
        required: true
        content:
          application/json:
            schema:
              type: object
              additionalProperties: false
              required: [transaction_id]
              properties:
                transaction_id: { type: string, minLength: 1, maxLength: 80 }
      responses:
        '200':
          description: Rules result plus optional Jev result.
          content:
            application/json:
              schema: { $ref: '#/components/schemas/Analysis' }
        '404': { $ref: '#/components/responses/NotFound' }
  /api/v1/alerts:
    get:
      operationId: listAlerts
      responses:
        '200':
          description: Alerts for the current demo user.
          content:
            application/json:
              schema:
                type: array
                items: { $ref: '#/components/schemas/Alert' }
  /api/v1/privacy/preview/{transaction_id}:
    get:
      operationId: previewPrivacyPayload
      parameters:
        - $ref: '#/components/parameters/TransactionId'
      responses:
        '200':
          description: Exact safe payload and transformation explanation.
          content:
            application/json:
              schema: { $ref: '#/components/schemas/PrivacyPreview' }
        '404': { $ref: '#/components/responses/NotFound' }
  /api/v1/alerts/{alert_id}/confirmations:
    post:
      operationId: confirmAlert
      parameters:
        - $ref: '#/components/parameters/AlertId'
      requestBody:
        required: true
        content:
          application/json:
            schema:
              type: object
              additionalProperties: false
              required: [decision, consent_to_memory]
              properties:
                decision:
                  type: string
                  enum: [normal, review]
                consent_to_memory:
                  type: boolean
      responses:
        '200':
          description: Confirmation outcome and memory write status.
          content:
            application/json:
              schema: { $ref: '#/components/schemas/Confirmation' }
        '404': { $ref: '#/components/responses/NotFound' }
        '409':
          description: Conflicting or duplicate confirmation.
          content:
            application/json:
              schema: { $ref: '#/components/schemas/ErrorEnvelope' }
  /api/v1/memories:
    get:
      operationId: listMemories
      responses:
        '200':
          description: Sanitized summaries only.
          content:
            application/json:
              schema:
                type: array
                items: { $ref: '#/components/schemas/MemorySummary' }
  /api/v1/memories/{memory_id}:
    delete:
      operationId: deleteMemory
      parameters:
        - name: memory_id
          in: path
          required: true
          schema: { type: string, minLength: 1, maxLength: 100 }
      responses:
        '204': { description: Deleted }
        '404': { $ref: '#/components/responses/NotFound' }
components:
  parameters:
    TransactionId:
      name: transaction_id
      in: path
      required: true
      schema: { type: string, minLength: 1, maxLength: 80 }
    AlertId:
      name: alert_id
      in: path
      required: true
      schema: { type: string, minLength: 1, maxLength: 80 }
  responses:
    NotFound:
      description: Resource is absent or is not owned by the current demo identity.
      content:
        application/json:
          schema: { $ref: '#/components/schemas/ErrorEnvelope' }
  schemas:
    HealthResponse:
      type: object
      additionalProperties: false
      required: [status, data_mode, analysis_mode, memory_mode, request_id]
      properties:
        status: { type: string, enum: [ok, degraded] }
        data_mode: { type: string, enum: [synthetic, nessie-live, degraded] }
        analysis_mode: { type: string, enum: [rules-only, rules+jev-live, degraded] }
        memory_mode: { type: string, enum: [off, local-fallback, backboard-live, degraded] }
        request_id: { type: string }
    Transaction:
      type: object
      additionalProperties: false
      required: [id, occurred_at, amount, currency, direction, merchant_display, merchant_category, source, account_token, is_subscription]
      properties:
        id: { type: string }
        occurred_at: { type: string, format: date-time }
        amount: { type: number, minimum: 0 }
        currency: { type: string, enum: [USD] }
        direction: { type: string, enum: [debit, credit] }
        merchant_display: { type: string, maxLength: 80 }
        merchant_category: { type: string, maxLength: 40 }
        source: { type: string, enum: [synthetic, nessie] }
        account_token: { type: string, pattern: '^acct_[A-Za-z0-9_-]+$' }
        is_subscription: { type: boolean }
    Analysis:
      type: object
      additionalProperties: false
      required: [transaction_id, alert_type, severity, needs_human, confidence, evidence, provider_mode, request_id]
      properties:
        transaction_id: { type: string }
        alert_type: { type: string, enum: [duplicate, amount_anomaly, subscription_change, none] }
        severity: { type: string, enum: [low, medium, high] }
        needs_human: { type: boolean }
        confidence: { type: number, minimum: 0, maximum: 1 }
        evidence:
          type: array
          maxItems: 8
          items: { type: string, maxLength: 180 }
        provider_mode: { type: string, enum: [rules-only, rules+jev-live, degraded] }
        model_version: { type: [string, 'null'] }
        request_id: { type: string }
    Alert:
      type: object
      additionalProperties: false
      required: [transaction_id, alert_type, severity, needs_human, confidence, evidence, provider_mode, request_id, alert_id, status]
      properties:
        transaction_id: { type: string }
        alert_type: { type: string, enum: [duplicate, amount_anomaly, subscription_change, none] }
        severity: { type: string, enum: [low, medium, high] }
        needs_human: { type: boolean }
        confidence: { type: number, minimum: 0, maximum: 1 }
        evidence:
          type: array
          maxItems: 8
          items: { type: string, maxLength: 180 }
        provider_mode: { type: string, enum: [rules-only, rules+jev-live, degraded] }
        model_version: { type: [string, 'null'] }
        request_id: { type: string }
        alert_id: { type: string }
        status: { type: string, enum: [open, normal, review] }
    FieldAction:
      type: object
      additionalProperties: false
      required: [field, action, reason]
      properties:
        field: { type: string }
        action: { type: string, enum: [allowed, dropped, tokenized, generalized, bucketed] }
        output: {}
        reason: { type: string, maxLength: 180 }
    PrivacyPreview:
      type: object
      additionalProperties: false
      required: [transaction_id, destination, schema_version, inspection, safe_payload]
      properties:
        transaction_id: { type: string }
        destination: { type: string, enum: [jev, backboard-memory] }
        schema_version: { type: string, enum: [safe-ai-payload-v1] }
        inspection:
          type: array
          items: { $ref: '#/components/schemas/FieldAction' }
        safe_payload:
          type: object
          additionalProperties: false
          required: [merchant_category, amount_bucket, currency, local_day_part, same_category_same_amount_24h_count, is_subscription, subscription_delta_bucket, prior_user_confirmation]
          properties:
            merchant_category: { type: string }
            amount_bucket: { type: string, enum: ['0-10', '10-25', '25-50', '50-100', '100-250', '250+'] }
            currency: { type: string, enum: [USD] }
            local_day_part: { type: string, enum: [morning, afternoon, evening, overnight] }
            same_category_same_amount_24h_count: { type: integer, minimum: 0, maximum: 20 }
            is_subscription: { type: boolean }
            subscription_delta_bucket: { type: string, enum: [none, up-to-20-percent, over-20-percent, unknown] }
            prior_user_confirmation: { type: string, enum: [normal, review, unknown] }
    Confirmation:
      type: object
      additionalProperties: false
      required: [alert_id, decision, memory_status, request_id]
      properties:
        alert_id: { type: string }
        decision: { type: string, enum: [normal, review] }
        memory_status: { type: string, enum: [not-requested, written, pending, skipped, failed] }
        memory_id: { type: [string, 'null'] }
        request_id: { type: string }
    MemorySummary:
      type: object
      additionalProperties: false
      required: [memory_id, summary, created_at, source]
      properties:
        memory_id: { type: string }
        summary: { type: string, maxLength: 240 }
        created_at: { type: string, format: date-time }
        source: { type: string, enum: [backboard, local-fallback] }
    ErrorEnvelope:
      type: object
      additionalProperties: false
      required: [error]
      properties:
        error:
          type: object
          additionalProperties: false
          required: [code, message, request_id]
          properties:
            code: { type: string }
            message: { type: string }
            request_id: { type: string }
